{{- /* Renders an image using the media registry so the published markup matches the responsive set the previous build produced. */ -}} {{- $src := .src -}} {{- $entry := index site.Data.media (strings.TrimPrefix "/" $src) -}} {{- if not $entry -}} {{- errorf "image not found in the media registry: %s" $src -}} {{- end -}} {{- $sizes := .sizes | default (index $entry "sizes") -}} {{- $loading := .loading | default "lazy" -}} {{- $class := .class | default "" -}} {{- $alt := .alt | default "" -}}
