{{- /* Places one of the site's figures in the page. The figure is read as data and inlined, so it is part of the document rather than a separate frame. That means the site's fonts, the site's colours and the site's colour mode all reach it. Usage: {{< embed "trade-off-frontier" >}} */ -}} {{- $name := or (.Get "name") (.Get 0) -}} {{- if not $name -}} {{- errorf "the embed shortcode needs the name of a figure" -}} {{- end -}} {{- $figure := resources.Get (printf "embeds/%s.html" $name) -}} {{- if not $figure -}} {{- errorf "no figure called %q in assets/embeds" $name -}} {{- end -}} {{- $figure.Content | safeHTML -}}
